def get_settings() -> Settings:
    return Settings()

Create Makefile with these targets:
```makefile
.PHONY: setup run clean doctor

setup:
	uv sync
	@echo "✓ Dependencies installed. Copy .env.local.example to .env.local and add your NVIDIA_API_KEY."

run:
	uv run streamlit run app.py

clean:
	rm -rf .venv __pycache__ core/__pycache__ data/uploads data/cache data/chroma
	@echo "✓ Cleaned."

doctor:
	@echo "Checking environment..."
	@uv run python -c "from core.config import get_settings; s = get_settings(); print(f'API Key: {s.nvidia_api_key[:8]}...'); print(f'Base URL: {s.nvidia_base_url}'); print(f'Model: {s.nvidia_model}'); print('✓ Config OK')"
```
