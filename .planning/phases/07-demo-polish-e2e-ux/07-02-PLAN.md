   function Invoke-Demo {
       Write-Host "=== DocBot E2E Demo Dry-Run ===" -ForegroundColor Cyan
       $streamlitProcess = Start-Process -FilePath "uv" -ArgumentList "run", "streamlit", "run", "app.py", "--server.headless", "true" -PassThru -WindowStyle Hidden
       Write-Host "  Waiting for Streamlit server..." -ForegroundColor Cyan
       $ready = $false
       for ($i = 0; $i -lt 30; $i++) {
           try {
               $response = Invoke-WebRequest -Uri "http://localhost:8501/_stcore/health" -UseBasicParsing -TimeoutSec 2 -ErrorAction Stop
               if ($response.StatusCode -eq 200) { $ready = $true; break }
           } catch { }
           Start-Sleep -Seconds 1
       }
       if (-not $ready) {
           Write-Host "✗ Streamlit server failed to start within 30s." -ForegroundColor Red
           if (-not $streamlitProcess.HasExited) { Stop-Process -Id $streamlitProcess.Id -Force }
           exit 1
       }
       Write-Host "  Streamlit server ready. Running E2E tests..." -ForegroundColor Green
       uv run pytest tests/test_e2e_demo.py -x -q
       $testExit = $LASTEXITCODE
       if (-not $streamlitProcess.HasExited) { Stop-Process -Id $streamlitProcess.Id -Force }
       if ($testExit -ne 0) {
           Write-Host "✗ E2E demo dry-run failed." -ForegroundColor Red
           exit 1
       }
       Write-Host "`n✓ E2E demo dry-run passed." -ForegroundColor Green
   }
   ```

def streamlit_server():
    """Start Streamlit server for E2E tests, kill after module completes."""
    proc = subprocess.Popen(
        ["uv", "run", "streamlit", "run", "app.py", "--server.headless", "true"],
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
    )
    # Wait for server ready
    import urllib.request
    for _ in range(SERVER_TIMEOUT):
        try:
            resp = urllib.request.urlopen(f"{STREAMLIT_URL}/_stcore/health", timeout=2)
            if resp.status == 200:
                break
        except Exception:
            pass
        time.sleep(1)
    else:
        proc.kill()
        pytest.fail("Streamlit server failed to start within timeout")
    yield proc
    proc.kill()
    proc.wait()
```

```python
def test_e2e_demo_flow(streamlit_server, page: Page):
    """Full demo flow: upload → chat → summarize → graph → compare."""
    page.goto(STREAMLIT_URL)
    page.wait_for_load_state("networkidle")

    # D-12a: Load sample document
    # Find and click the sample loader button (exact selector depends on Streamlit DOM)
    # Look for button text containing "Load" or "Sample" in the sidebar/main area
    # Use page.get_by_role or page.locator with Streamlit's data-testid attributes
    sample_button = page.get_by_role("button", name=re.compile(r"load|sample", re.IGNORECASE))
    sample_button.first.click()

    # Wait for processing to complete — look for success indicator or sidebar doc entry
    page.wait_for_timeout(15000)  # Allow time for upload + processing + embedding

    # D-12b: Verify document appears in sidebar
    # Streamlit renders sidebar content — check for document filename or "documents" text
    sidebar = page.locator('[data-testid="stSidebar"]')
    expect(sidebar).to_contain_text(re.compile(r"\.pdf|document", re.IGNORECASE), timeout=20000)

    # D-12c: Chat — switch to Chat tab and ask a question
    page.get_by_role("tab", name=re.compile("Chat")).click()
    chat_input = page.locator('textarea[data-testid="stChatInputTextArea"]')
    chat_input.fill("What is this document about?")
    chat_input.press("Enter")

    # Wait for response and verify non-empty answer
    page.wait_for_timeout(15000)  # Allow LLM response time
    # Look for assistant message in chat
    messages = page.locator('[data-testid="stChatMessage"]')
    expect(messages.last).to_be_visible(timeout=30000)
    response_text = messages.last.inner_text()
    assert len(response_text) > 10, f"Chat response too short: {response_text!r}"

    # D-12d: Summary — switch to Summary tab
    page.get_by_role("tab", name=re.compile("Summary")).click()
    page.wait_for_timeout(1000)
    # Click summarize button (exact text from summary_view.py)
    summarize_btn = page.get_by_role("button", name=re.compile(r"summar", re.IGNORECASE))
    summarize_btn.first.click()
    page.wait_for_timeout(20000)  # Map-reduce can be slow
    # Verify summary output exists (not empty, not error)
    summary_content = page.locator("main")
    expect(summary_content).to_contain_text(re.compile(r"\w{20,}"), timeout=30000)

    # D-12e: Graph — switch to Graph tab
    page.get_by_role("tab", name=re.compile("Graph")).click()
    page.wait_for_timeout(1000)
    extract_btn = page.get_by_role("button", name=re.compile(r"extract", re.IGNORECASE))
    extract_btn.first.click()
    page.wait_for_timeout(20000)  # Graph extraction + dedup
    # Verify entities rendered (table or graph component)
    graph_content = page.locator("main")
    expect(graph_content).to_contain_text(re.compile(r"entit|node|relationship", re.IGNORECASE), timeout=30000)

    # D-12f: Compare — switch to Compare tab
    page.get_by_role("tab", name=re.compile("Compare")).click()
    page.wait_for_timeout(1000)
    # Find compare input and button
    compare_input = page.locator('input[type="text"]').last
    compare_input.fill("What is this document about?")
    compare_btn = page.get_by_role("button", name=re.compile(r"compare", re.IGNORECASE))
    compare_btn.first.click()
    page.wait_for_timeout(20000)  # Parallel LLM calls
    # Verify two results appear (two columns with content)
    compare_content = page.locator("main")
    expect(compare_content).to_contain_text(re.compile(r"large|direct|route", re.IGNORECASE), timeout=30000)
```

**Important implementation notes:**
- Add `import re` at top of file
- The exact Streamlit selectors (data-testid attributes, role names) must be discovered by reading the actual rendered DOM. The selectors above use Playwright's recommended patterns (get_by_role, locator with data-testid) which work with Streamlit's component structure.
- If a selector doesn't match during execution, the executor should inspect the page DOM and adjust selectors accordingly.
- Timeouts are generous (15-30s) because NIM API calls take 5-15s each and the test runs against a live API.
- The `page` fixture is provided by pytest-playwright automatically (no explicit fixture needed).
- D-18 (sidebar updates after upload) is verified by the sidebar assertion after sample loading.
- D-16 (no page reloads) is verified by the sequential tab switching — if any tab required a reload, subsequent steps would fail.
- D-17 (error states) is verified implicitly — if errors surface instead of results, assertions fail.
